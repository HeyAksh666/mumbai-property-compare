loc_names  = list(LOCATIONS.keys())
loc_probs  = np.array([3,2,4,3,3,3,3,3,2,2,2,2,2,2,2,
                        4,4,3,3,3,2,2,2,2,1], dtype=float)
loc_probs /= loc_probs.sum()

locations  = np.random.choice(loc_names, size=N, p=loc_probs)

def make_row(loc):
    city, base_sale, base_rent = LOCATIONS[loc]

    prop_type   = np.random.choice(
        ["Apartment","Studio","Villa","Row House","Penthouse"],
        p=[0.60, 0.10, 0.12, 0.10, 0.08]
    )
    listing     = np.random.choice(["Sale", "Rent"], p=[0.55, 0.45])

    # BHK depends on property type
    if prop_type == "Studio":
        bhk = 1
    elif prop_type in ("Villa", "Row House"):
        bhk = np.random.choice([3, 4, 5], p=[0.4, 0.4, 0.2])
    elif prop_type == "Penthouse":
        bhk = np.random.choice([3, 4, 5], p=[0.3, 0.5, 0.2])
    else:
        bhk = np.random.choice([1, 2, 3, 4], p=[0.20, 0.40, 0.30, 0.10])

    bathrooms   = min(bhk + np.random.randint(0, 2), 6)
    area        = round(bhk * np.random.uniform(380, 620) + np.random.normal(0, 50), 1)
    area        = max(area, 250)

    floor       = np.random.randint(0, 35)
    tot_floors  = floor + np.random.randint(1, 15)
    age         = round(np.random.exponential(7), 1)

    furnishing  = np.random.choice(
        ["Unfurnished", "Semi-Furnished", "Fully Furnished"],
        p=[0.30, 0.40, 0.30]
    )
    parking     = np.random.choice([0, 1, 2], p=[0.25, 0.55, 0.20])
    lift        = 1 if tot_floors > 4 else int(np.random.rand() > 0.3)
    gym         = int(np.random.rand() > 0.55)
    pool        = int(np.random.rand() > 0.75)
    security    = int(np.random.rand() > 0.40)
    metro_dist  = round(np.random.exponential(1.2), 2)   # km to nearest metro
    school_dist = round(np.random.uniform(0.3, 3.5), 2)  # km to nearest school

    # ── Price calculation (realistic Mumbai logic) ────────────
    area_price  = base_sale * area
    bhk_mult    = 1 + (bhk - 2) * 0.12
    floor_bonus = 1 + floor * 0.003
    age_disc    = max(0.70, 1 - age * 0.015)
    furnish_m   = {"Unfurnished": 1.0, "Semi-Furnished": 1.05,
                   "Fully Furnished": 1.12}[furnishing]
    amenity_m   = 1 + 0.02*gym + 0.03*pool + 0.01*security + 0.02*lift
    metro_disc  = max(0.88, 1 - metro_dist * 0.04)
    type_m      = {"Apartment":1.0,"Studio":0.85,"Villa":1.35,
                   "Row House":1.20,"Penthouse":1.50}[prop_type]

    noise       = np.random.normal(1.0, 0.06)

    if listing == "Sale":
        price = (area_price * bhk_mult * floor_bonus * age_disc *
                 furnish_m * amenity_m * metro_disc * type_m * noise)
        price = round(price / 100000, 2)   # convert to Lakhs
    else:
        price = (base_rent * area/600 * bhk_mult * furnish_m *
                 amenity_m * metro_disc * type_m * noise * 1000)
        price = round(price / 1000) * 1000  # round to nearest 1000

    return {
        "location": loc, "city": city, "property_type": prop_type,
        "bhk": bhk, "bathrooms": bathrooms, "area_sqft": area,
        "floor": floor, "total_floors": tot_floors, "age_years": age,
        "furnishing": furnishing, "parking": parking, "lift": lift,
        "gym": gym, "pool": pool, "security": security,
        "metro_dist_km": metro_dist, "school_dist_km": school_dist,
        "listing_type": listing, "price": price
    }

rows = [make_row(loc) for loc in locations]
df   = pd.DataFrame(rows)

# ── Inject realistic missing values ──────────────────────────
for col, rate in [("age_years",0.05),("parking",0.03),

# Step 1 → Remove duplicates
df_clean.drop_duplicates(inplace=True)
print(f"\n✅ After dedup: {len(df_clean):,} rows")

# Step 2 → Outlier removal using IQR per listing_type
for lt in ["Sale", "Rent"]:
    mask = df_clean["listing_type"] == lt
    Q1   = df_clean.loc[mask, "price"].quantile(0.01)
    Q3   = df_clean.loc[mask, "price"].quantile(0.99)
    IQR  = Q3 - Q1
    lo   = Q1 - 3.0 * IQR
    hi   = Q3 + 3.0 * IQR
    before = mask.sum()
    bad    = (df_clean["listing_type"] == lt) & (
             (df_clean["price"] < lo) | (df_clean["price"] > hi))
    df_clean = df_clean[~bad]
    print(f"   Outliers removed ({lt}): {bad.sum()} rows")

# Step 3 → Outlier removal on area_sqft
area_q99 = df_clean["area_sqft"].quantile(0.99)
df_clean  = df_clean[df_clean["area_sqft"] <= area_q99]

# Step 4 → Impute missing values
num_cols = df_clean.select_dtypes(include=np.number).columns.tolist()
cat_cols = df_clean.select_dtypes(include="object").columns.tolist()

# Numeric → median (robust to skew)
for col in num_cols:
    if df_clean[col].isnull().any():
        med = df_clean[col].median()
        df_clean[col].fillna(med, inplace=True)
        print(f"   Imputed '{col}' with median={med:.2f}")

# Categorical → mode
for col in cat_cols:
    if df_clean[col].isnull().any():
        mode = df_clean[col].mode()[0]
        df_clean[col].fillna(mode, inplace=True)

# Step 5 → Logical consistency checks
df_clean = df_clean[df_clean["bathrooms"] <= df_clean["bhk"] + 2]

# ── 3. Area-based features ────────────────────────────────────
df_feat["area_sqft_sq"]   = df_feat["area_sqft"] ** 2          # non-linear area
df_feat["area_per_bhk"]   = df_feat["area_sqft"] / df_feat["bhk"]  # space per room
df_feat["area_bhk_inter"] = df_feat["area_sqft"] * df_feat["bhk"]  # interaction

# ── 4. Floor features ─────────────────────────────────────────
df_feat["floor_ratio"]    = df_feat["floor"] / (df_feat["total_floors"] + 1)
df_feat["is_top_floor"]   = (df_feat["floor"] == df_feat["total_floors"]).astype(int)
df_feat["is_ground_floor"]= (df_feat["floor"] == 0).astype(int)
df_feat["is_high_rise"]   = (df_feat["total_floors"] >= 20).astype(int)

# ── 5. Ordinal encoding — furnishing ─────────────────────────
furn_map = {"Unfurnished": 0, "Semi-Furnished": 1, "Fully Furnished": 2}
df_feat["furnishing_ord"] = df_feat["furnishing"].map(furn_map)

# ── 6. Amenity score (composite) ─────────────────────────────
df_feat["amenity_score"] = (df_feat["lift"]     * 1.0 +
                             df_feat["gym"]      * 1.5 +
                             df_feat["pool"]     * 2.0 +
                             df_feat["security"] * 0.5 +
                             df_feat["parking"]  * 1.0)

# ── 7. Location tier (Mumbai micro-market classification) ──────
premium_locs   = ["Worli","Bandra West","Lower Parel","Juhu","Dadar"]
mid_locs       = ["Andheri West","Powai","Goregaon West","Santacruz West",
                  "Malad West","Borivali West","Vashi","Nerul","Belapur"]

def get_tier(row):
    if row["location"] in premium_locs:   return 3   # Premium
    elif row["location"] in mid_locs:     return 2   # Mid-market
    elif row["city"] == "Mumbai":         return 1   # Suburban Mumbai
    else:                                 return 0   # Navi Mumbai

df_feat["location_tier"] = df_feat.apply(get_tier, axis=1)

# ── 8. Target encoding — location (mean price per location) ───
# IMPORTANT: Fit ONLY on training data to prevent leakage
loc_means = df_feat.groupby("location")["price"].mean()
df_feat["location_mean_price"] = df_feat["location"].map(loc_means)

# ── 9. Building age buckets ───────────────────────────────────
df_feat["age_bucket"] = pd.cut(df_feat["age_years"],
                                bins=[-1, 2, 7, 15, 100],
                                labels=["New(0-2y)","Mid(2-7y)",
                                        "Old(7-15y)","Very Old(15y+)"])
df_feat["age_bucket_ord"] = df_feat["age_bucket"].cat.codes

# ── 10. Property type dummies ─────────────────────────────────
df_feat = pd.get_dummies(df_feat, columns=["property_type"], prefix="ptype",
                          drop_first=False)
df_feat = pd.get_dummies(df_feat, columns=["city"], prefix="city",
                          drop_first=False)

# ── Summary ───────────────────────────────────────────────────
print(f"✅ Feature engineering complete.")
print(f"   Original features : 19")
print(f"   Engineered features: {df_feat.shape[1]}")
print(f"\n📋 New features created:")
new_feats = ["log_price","area_sqft_sq","area_per_bhk","floor_ratio",

             "location_tier","location_mean_price","age_bucket_ord"]

    print(f"   ✔ {f}")
